cmake_minimum_required(VERSION 3.20)
project(cet4_study LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Drogon CONFIG REQUIRED)
if(TARGET SQLite3_lib)
  set(CET4_SQLITE_TARGET SQLite3_lib)
elseif(TARGET SQLite::SQLite3)
  set(CET4_SQLITE_TARGET SQLite::SQLite3)
else()
  find_package(SQLite3 REQUIRED)
  set(CET4_SQLITE_TARGET SQLite::SQLite3)
endif()
add_executable(cet4 backend/main.cpp)
target_link_libraries(cet4 PRIVATE Drogon::Drogon ${CET4_SQLITE_TARGET})
if(MSVC)
  target_compile_options(cet4 PRIVATE /utf-8)
endif()
